f_price(float p) =>
    na(p) ? "n/a" :
     p >= 10000 ? "$" + str.tostring(math.round(p, 0)) :
     p >= 1000  ? "$" + str.tostring(math.round(p, 0)) :
     p >= 100   ? "$" + str.tostring(math.round(p, 1)) :
     p >= 10    ? "$" + str.tostring(math.round(p, 2)) :
     p >= 1     ? "$" + str.tostring(math.round(p, 3)) :
     p >= 0.01  ? "$" + str.tostring(math.round(p, 4)) :
     p >= 0.0001? "$" + str.tostring(math.round(p, 6)) :
                  "$" + str.tostring(math.round(p, 8))

f_status(float price, float lo, float hi) =>
    string r = ""
    if na(price)
        r := "BAD SYMBOL"
    else if price >= lo and price <= hi
        float pos = (price - lo) / (hi - lo) * 100
        r := "IN (" + str.tostring(math.round(pos, 0)) + "%)"
    else if price > hi
        float pct = (price - hi) / hi * 100
        if pct < 2.0
            r := "AT ZONE (+" + str.tostring(math.round(pct, 1)) + "%)"
        else
            r := "+" + str.tostring(math.round(pct, 1)) + "% above"
    else
        float pct = (lo - price) / price * 100
        r := str.tostring(math.round(pct, 1)) + "% to go"
    r

// ─────────────────────────────────────────────
// SORT KEY — bucketed T1 ranking, top -> bottom within each tier
// ─────────────────────────────────────────────
f_sortkey(float price, float lo, float hi) =>
    float BELOW_BASE = -1000.0
    float IN_BASE    = 0.0
    float AT_BASE    = 1000.0
    float GO_BASE    = 2000.0
    float NA_BASE    = 9.0e8
    float k = NA_BASE
    if na(price)
        k := NA_BASE
    else if price >= lo and price <= hi
        k := IN_BASE + (price - lo) / (hi - lo)
    else if price < lo
        k := BELOW_BASE - ((lo - price) / price * 100)
    else
        float pct = (price - hi) / hi * 100
        if pct < 2.0
            k := AT_BASE + pct
        else
            k := GO_BASE + pct
    k

f_bg(float price, float lo, float hi) =>
    color r = C_ABOVE_BG
    if na(price)
        r := color.new(#3a1515, 50)
    else if price >= lo and price <= hi
        r := C_IN_BG
    else if price > hi
        float pct = (price - hi) / hi * 100
        if pct < 2.0
            r := C_AT_BG
        else if pct < 8.0
            r := C_CLOSE_BG
        else
            r := C_ABOVE_BG
    r

f_tc(float price, float lo, float hi) =>
    color r = C_TXT_DIM
    if na(price)
        r := color.new(#ff4444, 0)
    else if price >= lo and price <= hi
        r := C_IN_TXT
    else if price > hi
        float pct = (price - hi) / hi * 100
        if pct < 2.0
            r := C_IN_TXT
        else if pct < 8.0
            r := C_CLOSE_TXT
        else
            r := C_TXT_DIM
    r

f_high_away(float px, float hi) =>
    na(px) or na(hi) or hi <= 0 ? "n/a" :
     px >= hi ? "AT HIGH" :
     str.tostring(math.round((hi - px) / hi * 100, 1)) + "% away"

f_high_tc(float px, float hi) =>
    na(px) or na(hi) ? C_TXT_DIM :
     px >= hi ? C_IN_TXT :
     (hi - px) / hi * 100 <= 5 ? C_CLOSE_TXT : C_TXT_DIM

f_low_away(float px, float lo) =>
    na(px) or na(lo) or lo <= 0 ? "n/a" :
     px <= lo ? "AT 2Y LOW" :
     "+" + str.tostring(math.round((px - lo) / lo * 100, 1)) + "%"

f_low_tc(float px, float lo) =>
    na(px) or na(lo) ? C_TXT_DIM :
     px <= lo ? C_IN_TXT :
     (px - lo) / lo * 100 <= 15 ? C_CLOSE_TXT :
     C_TXT_DIM

f_res_str(float price, float thresh) =>
    thresh <= 0 ? "—" :
     price < thresh ? "★ BELOW" :
     "+" + str.tostring(math.round((price - thresh) / thresh * 100, 1)) + "%"

f_res_bg(float price, float thresh) =>
    thresh <= 0 ? C_BG : price < thresh ? C_IN_BG : color.new(#21262d, 0)

f_res_tc(float price, float thresh) =>
    thresh <= 0 ? C_TXT_DIM : price < thresh ? C_IN_TXT : C_RES

f_action(string nm, float price, float t1l, float t1h, float t2l, float t2h, float t3l, float t3h) =>
    string r = ""
    if not na(price)
        if price >= t1l and price <= t1h
            r := nm + "(T1) "
        else if price > t1h and (price - t1h) / t1h * 100 < 2.0
            r := nm + "(~T1) "
        else if price >= t2l and price <= t2h
            r := nm + "(T2) "
        else if price > t2h and (price - t2h) / t2h * 100 < 2.0
            r := nm + "(~T2) "
        else if price >= t3l and price <= t3h
            r := nm + "(T3) "
        else if price > t3h and (price - t3h) / t3h * 100 < 2.0
            r := nm + "(~T3) "
    r

// ─────────────────────────────────────────────
flashPhase = flash_on and (math.floor(timenow / flash_ms) % 2 == 0)

f_inzone(float price, float lo, float hi) =>
    not na(price) and price >= lo and price <= hi

f_bg_flash(float price, float lo, float hi) =>
    color r = f_bg(price, lo, hi)
    if f_inzone(price, lo, hi)
        r := flashPhase ? color.new(#1a7f37, 0) : color.new(#1a7f37, 80)
    r

f_tc_flash(float price, float lo, float hi) =>
    color r = f_tc(price, lo, hi)
    if f_inzone(price, lo, hi)
        r := flashPhase ? color.new(#ffffff, 0) : color.new(#3fb950, 0)
    r

f_sect(table t, int r, int cols, string lbl, color sbg, color stc) =>
    table.cell(t, 0, r, lbl, text_color=stc, bgcolor=sbg, text_size=size.small, text_halign=text.align_left)
    for c = 1 to cols - 1
        table.cell(t, c, r, "", text_color=stc, bgcolor=sbg, text_size=size.tiny)

f_emit_tier(table t, RowBuf b, int start_row, bool s_res, bool do_sort) =>
    int n = array.size(b.nm)
    var_order = array.new<int>()
    array.clear(var_order)
    for i = 0 to n - 1
        array.push(var_order, i)
    if do_sort
        for a = 0 to n - 2
            int minpos = a
            for bx = a + 1 to n - 1
                if array.get(b.key, array.get(var_order, bx)) < array.get(b.key, array.get(var_order, minpos))
                    minpos := bx
            if minpos != a
                int tmp = array.get(var_order, a)
                array.set(var_order, a, array.get(var_order, minpos))
                array.set(var_order, minpos, tmp)
    int rr = start_row
    for j = 0 to n - 1
        int idx = array.get(var_order, j)
        f_row(t, rr, s_res, array.get(b.nm, idx), array.get(b.px, idx), array.get(b.h52, idx),
              array.get(b.ath, idx), array.get(b.low2y, idx), array.get(b.nmtc, idx),
              array.get(b.t1l, idx), array.get(b.t1h, idx),
              array.get(b.t2l, idx), array.get(b.t2h, idx),
              array.get(b.t3l, idx), array.get(b.t3h, idx),
              array.get(b.res, idx))
        rr += 1
    rr
